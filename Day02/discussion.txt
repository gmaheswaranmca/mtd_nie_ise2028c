            name : ex1_server_sg 
            description: ex1_server_sg

                <Add rule> Custom TCP 8000 AnywhereIPv4 
                <Add rule> SSH 22 AnywhereIPv4
            Outbound:
                <Default> AllTraffic AnywhereIPv4
            <Create Security Group>
